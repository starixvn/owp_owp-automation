Verify Added or Edited Note
    [Arguments]    ${expected_content}
    ${status}    Run Keyword And Return Status    Wait Until Element Is Visible    xpath=//*[@id="root"]//li/div[@class='content']
    Run keyword if    '${status}'== 'True'    Verirify Note    ${expected_content}
    ...    ELSE    Log    ${expected_content} is deleted successfully
    Sleep    2s

Edit Note
    [Arguments]    ${orginal_note}    ${new_note}    ${action}
    ${content_note}    Set Variable    @${user_login}${SPACE}${orginal_note}
    ${status}    Run Keyword And Return Status    Wait Until Element Is Visible    xpath=//*[@id="root"]//li/div[@class='content']    timeout=20s
    ${count}    Run Keyword If    '${status}'== 'True'    Get Matching Xpath Count    xpath=//*[@id="root"]//li/div[@class='content']
    : FOR    ${i}    IN RANGE    1    ${count} + 1
    \    ${get_content}    get text    xpath=//*[@id="root"]//li[${i}]/div[@class='content']
    \    Run Keyword If    '${get_content}'=='${content_note}' and '${action}'=='Save'    Run Keywords    Edit Note_Expected Location    ${i}    ${new_note}
    \    ...    ${action}
    \    ...    AND    Exit For Loop
    \    Run Keyword If    '${get_content}'=='${content_note}' and '${action}'=='Cancel'    Run Keywords    Edit Note_Expected Location    ${i}    ${new_note}
    \    ...    ${action}
    \    ...    AND    Exit For Loop
    Sleep    3s

Edit Note_Expected Location
    [Arguments]    ${i}    ${new_note}    ${action}

Delete Note

    ${content_note}    Set Variable    @${user_login}${SPACE}${orginal_note}
    ${status}    Run Keyword And Return Status    Wait Until Element Is Visible    xpath=//*[@id="root"]//li/div[@class='content']    timeout=60s
    ${count}    Run Keyword If    '${status}'== 'True'    Get Matching Xpath Count    xpath=//*[@id="root"]//li/div[@class='content']
    : FOR    ${i}    IN RANGE    1    ${count} + 1
    \    ${get_content}    get text    xpath=//*[@id="root"]//li[${i}]/div[@class='content']
    \    Run Keyword If    '${get_content}' == '${content_note}' and '${action}' == 'Ok'    Run Keywords    Click Element    xpath=//*[@id="root"]//li[${i}]//button[@class='icon-delete']
    \    ...    AND    Get Alert and confirm    Are you sure you want to permanently delete this note?    ${action}
    \    ...    AND    Exit For Loop
    \    Run Keyword If    '${get_content}' == '${content_note}' and '${action}' == 'Cancel'    Run Keywords    Click Element    xpath=//*[@id="root"]//li[${i}]//button[@class='icon-delete']
    \    ...    AND    Get Alert and confirm    Are you sure you want to permanently delete this note?    ${action}
    \    ...    AND    Exit For Loop
    Sleep    3s

Verirify Note
    [Arguments]    ${expected_content}
    ${count}    Get Matching Xpath Count    xpath=//*[@id="root"]//li/div[@class='content']
    : FOR    ${i}    IN RANGE    1    ${count} + 1
    \    ${get_content}    Get Text    xpath=//*[@id="root"]//li[${i}]/div[@class='content']
    \    Log    ${get_content}
    \    Run Keyword If    '${get_content}'=='@${user_login}${SPACE}${expected_content}'    Run keywords    Should Be Equal    ${get_content}    @${user_login}${SPACE}${expected_content}
    \    ...    AND    Log    ${expected_content} is added or editted successfully
    \    ...    AND    Exit For Loop
    \    Run Keyword If    '${get_content}'!='@${user_login}${SPACE}${expected_content}'    Log    ${expected_content} has been deleted
    Sleep    2s
